{# ════════════════════════════════════════════════════════════
   اختبار شامل لـ Type Error و Scope Error و Division By Zero في Jinja2
   ════════════════════════════════════════════════════════════ #}

<html>
<body>

{# ── 1. أخطاء رؤى (TypeError) ── #}
{% for i in 5 %}          {# TypeError: 'int' object is not iterable #}
    <p>{{ i }}</p>
{% endfor %}

<p>{{ 5 + "hello" }}</p>  {# TypeError: unsupported operand type(s) for +: 'int' and 'str' #}

<p>{{ 10[0] }}</p>        {# TypeError: 'int' object is not subscriptable #}

<p>{{ "Sara" - 4 }}</p>   {# TypeError: unsupported operand type(s) for -: 'str' and 'int' #}


{# ── 2. أخطاء رغد (ScopeError) ── #}
{% for item in [1, 2, 3] %}
    <p>{{ item }}</p>
{% endfor %}
<p>{{ item }}</p>         {# UnboundLocalError: 'item' is not defined outside loop #}


{# ── 3. أخطاء رغد (DivisionByZeroError) ── #}
<p>{{ 10 / 0 }}</p>       {# ZeroDivisionError: division by zero #}
<p>{{ 7 % 0 }}</p>        {# ZeroDivisionError: integer division or modulo by zero #}


{# ── 4. حالات صحيحة (لازم ما تطلع أي خطأ - للتأكد من عدم الـ False Positives) ── #}
<p>{{ 10 / 2 }}</p>       {# ✅ صحيح (قسمة طبيعية) #}
<p>{{ 5 + 5 }}</p>        {# ✅ صحيح (جمع طبيعي) #}

{% set msg = "Hello" %}
<p>{{ msg }}</p>          {# ✅ صحيح (المتغير معرف بـ set ومستخدم بنفس السكوب) #}

</body>
</html>
