from flask import Flask, render_template

app = Flask(__name__)


@app.route("/")
def home():
    # ==========================================================
    # اختبار شامل لكل أخطاء Semantic Analysis في Python
    # ==========================================================

    # ─────────────────────────────────────────────
    # 1. أخطاء رؤى (TypeError)
    # ─────────────────────────────────────────────

    # Test 1: التكرار على نوع غير قابل للتكرار
    # المتوقع: TypeError: 'int' object is not iterable
    x1 = 10
    for i in x1:
        print(i)

    # Test 2: len() على نوع لا يدعم الطول
    # المتوقع: TypeError: object of type 'int' has no len()
    x2 = 10
    print(len(x2))

    # Test 3: فهرسة نوع غير قابل للفهرسة
    # المتوقع: TypeError: 'int' object is not subscriptable
    x3 = 5
    print(x3[0])

    # Test 4: جمع (+) بين أنواع غير متوافقة
    # المتوقع: TypeError: can only concatenate str (not "int") to str
    x4 = "Sara" + 4

    # Test 5: طرح (-) بين أنواع غير متوافقة
    # المتوقع: TypeError: unsupported operand type(s) for -: 'str' and 'int'
    x5 = "Sara" - 4

    # Test 6: طرح (-) بين نصين
    # المتوقع: TypeError: unsupported operand type(s) for -: 'str' and 'str'
    x6 = "Sara" - "Ali"

    # Test 7: مقارنة بين أنواع غير متوافقة
    # المتوقع: TypeError: '<' not supported between instances of 'str' and 'int'
    x7 = "Sara" < 4

    # Test 8: عملية حسابية مع None
    # المتوقع: TypeError: unsupported operand type(s) for +: 'NoneType' and 'int'
    x8 = None
    x8_result = x8 + 5

    # ─────────────────────────────────────────────
    # 2. أخطاء رغد (ScopeError)
    # ─────────────────────────────────────────────

    # Test 9: استخدام متغير غير معرف إطلاقاً (NameError)
    # المتوقع: NameError: name 'secret' is not defined
    print(secret)

    # Test 10: متغير معرف داخل دالة ومستخدم خارجها (ScopeError)
    # المتوقع: UnboundLocalError أو NameError
    def calculate():
        internal_var = 100
        return internal_var

    print(internal_var)  # ← خطأ Scope! internal_var معرفة داخل calculate فقط

    # ─────────────────────────────────────────────
    # 3. أخطاء رغد (DivisionByZeroError)
    # ─────────────────────────────────────────────

    # Test 11: قسمة على صفر مباشرة
    # المتوقع: ZeroDivisionError: division by zero
    y = 0
    z = 10 / y

    # Test 12: modulo على صفر
    # المتوقع: ZeroDivisionError: integer division or modulo by zero
    m = 10 % 0

    # ─────────────────────────────────────────────
    # 4. أخطاء راما (WrongArguments, InvalidFunctionCall, UseBeforeAssignment)
    # ─────────────────────────────────────────────

    # Test 13: Wrong Arguments Count (استدعاء دالة بعدد وسائط خاطئ)
    # المتوقع: TypeError: greet() takes 2 positional arguments but 3 were given
    def greet(name, msg):
        print(name, msg)

    greet("Ali", "Hi", "!")   # ← خطأ! 3 وسائط بدل 2
    # greet("Ali")            # (معلق لتجنب تعارض العدد الناقص مع الفحص)
    greet("Ali", "Hi")        # ✅ صحيح (2 وسائط)

    # Test 14: Invalid Function Call (استدعاء غير قابل للاستدعاء)
    # المتوقع: TypeError: 'int' object is not callable
    callable_var = 5
    callable_var()            # ← خطأ! int ليس قابل للاستدعاء

    str_var = "Ali"
    str_var()                 # ← خطأ! str ليس قابل للاستدعاء

    # Test 15: Use Before Assignment (استخدام قبل الإسناد)
    # المتوقع: UnboundLocalError: local variable 'val' referenced before assignment
    def use_before_assign():
        print(val)            # ← خطأ! استخدام قبل الإسناد
        val = 5
    use_before_assign()

    # Test 16: Return Type Mismatch (Bonus)
    # المتوقع: TypeError: return type expected int, got str
    def bar(flag):
        if flag:
            return 1
        return "hi"           # ← خطأ! الدالة ترجع int في طريق و str في طريق آخر

    def safe(flag):
        if flag:
            return 1
        return 2              # ✅ صحيح (كلاهما int)

    # ─────────────────────────────────────────────
    # 5. حالات صحيحة (لازم ما تطلع أي خطأ - للتأكد من عدم الـ False Positives)
    # ─────────────────────────────────────────────

    # السكوب: i بـ بايثون بتبقى متاحة بعد الـ for
    for i in range(5):
        x = 1
    print(i)                  # ✅ صحيح (ما فيه خطأ)

    # القسمة: قسمة على متغير مش معروف القيمة
    unknown_val = int("10")
    result = 5 / unknown_val   # ✅ صحيح (القيمة UNKNOWN وقت الـ compile)

    # ─────────────────────────────────────────────
    # 6. أخطاء رؤى (TypeMismatchError) - Flask Bridge
    # ─────────────────────────────────────────────

    # تمرير متغيرات للقالب ليتم فحص الفلاتر
    age = 22           # int
    name = "Sara"      # str
    price = 99.5       # float
    items = [1, 2, 3]  # list

        # ─────────────────────────────────────────────
        # 7. أخطاء شهد Invalid Attribute Access (AttributeError)
        # ─────────────────────────────────────────────

        # Test 17: استدعاء method غير موجودة على str
        # المتوقع: AttributeError: 'str' object has no attribute 'push'
        attr_x1 = "hello"
        attr_x1.push("!")

        # Test 18: استدعاء method نصية على int
        # المتوقع: AttributeError: 'int' object has no attribute 'upper'
        attr_y1 = 5
        attr_y1.upper()

        # Test 19: استدعاء method على None
        # المتوقع: AttributeError: 'NoneType' object has no attribute 'upper'
        attr_z1 = None
        attr_z1.upper()

        # Test 20: استدعاء method خاطئة على list (الصح append مو add)
        # المتوقع: AttributeError: 'list' object has no attribute 'add'
        attr_lst1 = [1, 2, 3]
        attr_lst1.add(4)

        # ─────────────────────────────────────────────
        # 8. شهد: حالات صحيحة لنفس الـ Checker (يجب ألا تطلع أي AttributeError)
        # ─────────────────────────────────────────────

        # Test 21: استدعاء method صحيحة على str
        ok_str1 = "hello"
        ok_str1.upper()

        # Test 22: استدعاء method صحيحة على list
        ok_list1 = [1, 2, 3]
        ok_list1.append(4)

        # Test 23: استدعاء method صحيحة على dict
        ok_dict1 = {"a": 1}
        ok_dict1.get("a")

        # Test 24: chaining صحيح (str → str → method نصية أخرى)
        ok_chain1 = "hello"
        ok_chain1.strip().upper()

        # Test 25: attribute access على نوع غير معروف وقت الـ compile (لازم ما يطلع خطأ)
        ok_unknown1 = int("10")
        ok_unknown1.bit_length()

            # Test 26: attribute access جوا شرط if
            attr_cond = "hi"
            if attr_cond.foo():
                pass

            # Test 27: attribute access كـ argument لدالة تانية
            attr_arg = 5
            print(attr_arg.bar())

            # Test 28: attribute access على نتيجة استدعاء دالة (يجب ألا يطلع خطأ - UNKNOWN)
            unknown_result = greet("a", "b")
            unknown_result.whatever()

            # Test 29: attribute صحيحة داخل return statement جوا دالة
            def get_upper(s):
                return s.upper()   # ✅ لازم يعدي بدون خطأ لو s من نوع str معروف

            # Test 30: attribute access متسلسل بعد method ما بترجع str (false negative متوقع ومقبول)
            attr_chain2 = [1, 2, 3]
            attr_chain2.append(4).sort()   # نظريًا append بترجع None فعليًا، بس نحن ما بنتتبعها فبتعدي بصمت (سلوك متوقع ومقصود حسب فلسفة "False Positive أسوأ")


    return render_template(
        "htmlTestErrors.html",
        age=age,
        name=name,
        price=price,
        items=items
    )
