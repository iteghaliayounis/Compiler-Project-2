from flask import Flask, render_template

app = Flask(__name__)


@app.route("/")
def home():
    # ==========================================================
    # اختبار شامل لكل أخطاء Semantic Analysis في Python
    # ==========================================================

    # ─────────────────────────────────────────────
    # 1. أخطاء رؤى (TypeError)
    # ─────────────────────────────────────────────

    # Test 1: التكرار على نوع غير قابل للتكرار
    # المتوقع: TypeError: 'int' object is not iterable
    x1 = 10
    for i in x1:
        print(i)

    # Test 2: len() على نوع لا يدعم الطول
    # المتوقع: TypeError: object of type 'int' has no len()
    x2 = 10
    print(len(x2))

    # Test 3: فهرسة نوع غير قابل للفهرسة
    # المتوقع: TypeError: 'int' object is not subscriptable
    x3 = 5
    print(x3[0])

    # Test 4: جمع (+) بين أنواع غير متوافقة
    # المتوقع: TypeError: can only concatenate str (not "int") to str
    x4 = "Sara" + 4

    # Test 5: طرح (-) بين أنواع غير متوافقة
    # المتوقع: TypeError: unsupported operand type(s) for -: 'str' and 'int'
    x5 = "Sara" - 4

    # Test 6: طرح (-) بين نصين
    # المتوقع: TypeError: unsupported operand type(s) for -: 'str' and 'str'
    x6 = "Sara" - "Ali"

    # Test 7: مقارنة بين أنواع غير متوافقة
    # المتوقع: TypeError: '<' not supported between instances of 'str' and 'int'
    x7 = "Sara" < 4

    # Test 8: عملية حسابية مع None
    # المتوقع: TypeError: unsupported operand type(s) for +: 'NoneType' and 'int'
    x8 = None
    x8_result = x8 + 5

    # ─────────────────────────────────────────────
    # 2. أخطاء رغد (ScopeError)
    # ─────────────────────────────────────────────

    # Test 9: استخدام متغير غير معرف
    # المتوقع: NameError أو UnboundLocalError
    print(secret)

 # ════════════════════════════════════════════════════════════
    # 2. أخطاء رغد (ScopeError)
    # ════════════════════════════════════════════════════════════


    print(secret)          # UnboundLocalError: local variable 'secret' referenced before assignment


    # ════════════════════════════════════════════════════════════
    # 3. أخطاء رغد (DivisionByZeroError)
    # ════════════════════════════════════════════════════════════
    y = 0
    z = 10 / y             # ZeroDivisionError: division by zero

    m = 10 % 0             # ZeroDivisionError: integer division or modulo by zero


    # ════════════════════════════════════════════════════════════
    # 4. حالات صحيحة (لازم ما تطلع أي خطأ - للتأكد من عدم الـ False Positives)

    # السكوب: i بـ بايثون بتبقى متاحة بعد الـ for
    for i in range(5):
        pass
    print(i)               # ✅ صحيح (ما فيه خطأ)

    # القسمة: قسمة على متغير مش معروف القيمة
    unknown_val = int("10")
    result = 5 / unknown_val  # ✅ صحيح (ما فيه خطأ - القيمة UNKNOWN وقت الـ compile)

    return render_template("home.html")


    # ─────────────────────────────────────────────
    # 5. أخطاء رؤى (TypeMismatchError) - Flask Bridge
    # ─────────────────────────────────────────────

    # تمرير متغيرات للقالب ليتم فحص الفلاتر
    age = 22           # int
    name = "Sara"      # str
    price = 99.5       # float
    items = [1, 2, 3]  # list

    return render_template(
        "htmlTestErrors.html",
        age=age,
